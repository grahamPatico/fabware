# CAD IR Phase 12 — Inline-part fabrication cost estimation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development.

**Goal:** Estimate the material cost of fabricating each inline part from its CAD IR (volume × density × $/kg), then add it to the BOM/cost output. The agent gets a real total cost — externals + custom-cut parts — and the budget rule fires against the full number.

**Why this slice:** Phase 9 added BOM, Phase 10 added cost — but only for off-the-shelf externals. Custom-cut inline parts (the bulk of an assembly's mass for many designs) currently contribute $0 to the cost total. Phase 12 closes that loop with a simple material-volume model.

**Limitations declared up-front:**
- Volume estimator handles `extrude` (rect/circle sketches) and `cut_extrude` only. Other features (fillet/chamfer/hole/pattern/sweep/loft/shell/revolve/bend_flange/weld_tab) are ignored — produces a conservative overestimate (safer for budgeting).
- Machine time / setup fees / finishing surcharges are not included. Material-only.
- Material defaults to aluminum-6061 if not declared.

```ts
// convex/cad/compile/volume.ts
import type { CadIr } from "../ir/types";
import { resolveIr } from "../resolve/resolveIr";

const MM3_PER_CM3 = 1000;

function sketchAreaMm2(sketch: { geometry: Array<{ kind: string; width?: number; height?: number; radius?: number }> }): number {
  // For Phase 12 v0: support rect (w·h) and circle (π·r²). Other entities → 0.
  let area = 0;
  for (const g of sketch.geometry) {
    if (g.kind === "rect" && g.width !== undefined && g.height !== undefined) {
      area += g.width * g.height;
    } else if (g.kind === "circle" && g.radius !== undefined) {
      area += Math.PI * g.radius * g.radius;
    }
  }
  return area;
}

export function estimatePartVolumeCm3(ir: CadIr): number {
  let resolved;
  try {
    resolved = resolveIr(ir);
  } catch {
    return 0;
  }

  let volumeMm3 = 0;
  for (const f of resolved.features) {
    if (f.suppressed) continue;
    if (f.kind === "extrude") {
      const sketch = resolved.sketches[f.profile];
      if (!sketch) continue;
      volumeMm3 += sketchAreaMm2(sketch as never) * (f.distance as number);
    } else if (f.kind === "cut_extrude") {
      const sketch = resolved.sketches[f.profile];
      if (!sketch) continue;
      volumeMm3 -= sketchAreaMm2(sketch as never) * (f.distance as number);
    }
    // Other features ignored — conservative overestimate
  }

  return Math.max(0, volumeMm3 / MM3_PER_CM3);
}
```

```ts
// convex/cad/compile/fabricationCost.ts
import type { CadIr } from "../ir/types";
import { lookupMaterial, type Material, BUILTIN_MATERIALS } from "./materials";
import { estimatePartVolumeCm3 } from "./volume";

export interface FabricationCostEntry {
  partId: string;
  material: string;
  volumeCm3: number;
  weightKg: number;
  materialUsd: number;
}

export interface FabricationCostResult {
  totalUsd: number;
  perPart: FabricationCostEntry[];
}

export function compileFabricationCost(
  ir: CadIr,
  catalog: Record<string, Material> = BUILTIN_MATERIALS,
): FabricationCostResult {
  const perPart: FabricationCostEntry[] = [];
  let totalUsd = 0;

  if (ir.parts) {
    for (const [partId, part] of Object.entries(ir.parts)) {
      // Skip externals (BOM/cost already covers them)
      if ("kind" in part && part.kind === "external") continue;

      const partIr = (part as { ir: CadIr }).ir;
      const volumeCm3 = estimatePartVolumeCm3(partIr);
      if (volumeCm3 <= 0) continue;

      const material = lookupMaterial(partIr.material, catalog);
      const weightKg = (volumeCm3 * material.densityGramsPerCm3) / 1000;
      const materialUsd = weightKg * material.usdPerKg;

      perPart.push({
        partId,
        material: material.name,
        volumeCm3,
        weightKg,
        materialUsd,
      });
      totalUsd += materialUsd;
    }
  }

  // Top-level CadIr (no parts wrapper) also counts: walk the root part's features.
  if (!ir.parts || Object.keys(ir.parts).length === 0) {
    const volumeCm3 = estimatePartVolumeCm3(ir);
    if (volumeCm3 > 0) {
      const material = lookupMaterial(ir.material, catalog);
      const weightKg = (volumeCm3 * material.densityGramsPerCm3) / 1000;
      const materialUsd = weightKg * material.usdPerKg;
      perPart.push({ partId: "root", material: material.name, volumeCm3, weightKg, materialUsd });
      totalUsd += materialUsd;
    }
  }

  return { totalUsd, perPart };
}
```

```ts
// convex/cad/compile/cost.ts
import type { CadIr } from "../ir/types";
import { compileBom, type BomEntry } from "./bom";
import { compileFabricationCost, type FabricationCostEntry } from "./fabricationCost";

// existing pricing types and BUILTIN_PRICING stay

export interface CostResult {
  totalUsd: number;
  entries: CostEntry[];
  unknown: BomEntry[];
  fabrication: FabricationCostEntry[];
}

export function compileCost(ir: CadIr, db: PricingDb = BUILTIN_PRICING): CostResult {
  const bom = compileBom(ir);
  const entries: CostEntry[] = [];
  const unknown: BomEntry[] = [];
  let totalUsd = 0;

  for (const item of bom) {
    const key = `${item.vendor}::${item.partNumber}`;
    const unitUsd = db[key];
    if (unitUsd === undefined) {
      unknown.push(item);
      continue;
    }
    const lineTotalUsd = unitUsd * item.quantity;
    entries.push({ ...item, unitUsd, lineTotalUsd });
    totalUsd += lineTotalUsd;
  }

  const fab = compileFabricationCost(ir);
  totalUsd += fab.totalUsd;

  return { totalUsd, entries, unknown, fabrication: fab.perPart };
}
```

## Out of scope for Phase 12 (deferred)

- **Machine time / setup fees / finishing surcharges** — needs process model (laser-cut $/perimeter mm, CNC $/cm³ removed, 3D-print $/hour). Future phase.
- **Volume from sweep/loft/revolve/shell/fillet/chamfer/etc.** — Phase 12 v0 ignores these, which conservatively overestimates. Future phase: per-feature volume contributions.
- **Real vendor pricing for materials** — built-in $/kg values are representative, not sourced from a live API.
- **Part-by-part material override (per-feature material)** — current model is one-material-per-part. Multi-material parts would need per-feature material tags.
