# CAD IR Phase 13 — Process-aware machine-time cost

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development.

**Goal:** Extend Phase 12's material-only fabrication cost with process-specific machine-time charges. The agent declares an inline part's process (`laser_cut`, `cnc`, `print_3d`, `sheet_metal_bend`, or `none`); the cost compiler adds rate-based time/setup charges to the existing material cost. Total cost now reflects what a fabricator would actually quote.

**Why this slice:** Phase 12 ships fabrication material cost — but raw material is often the cheap part. A laser-cut bracket's bill is dominated by per-meter cutting cost and setup; a CNC machined block by volume-removed × tool-time; a 3D-print by build hours. Phase 13 adds those process-specific cost terms.

**Limitations:**
- Process detection is **declared, not inferred**. The agent picks `process` per part; absent → `none` (material-only, same as Phase 12).
- Estimators are simple: perimeter for laser, projected-area-volume for CNC, total volume for print. No tool-path planning, no nesting, no multi-pass logic.
- Rates are representative ($/m for laser, $/cm³ for CNC, $/cm³ for print, $/bend for sheet-metal). Caller can override at the boundary.
- No quote-grade pricing — accuracy is "right order of magnitude" for budget hints.

```ts
  usdPerCm3Removed: number;
  /** Per-cm³ build cost (3D print, additive volume). 0 for non-additive processes. */
  usdPerCm3Built: number;
  /** Per-bend cost (sheet-metal bending). 0 for non-bending processes. */
```

```ts
// convex/cad/compile/perimeter.ts
import type { CadIr } from "../ir/types";
import { resolveIr } from "../resolve/resolveIr";

const MM_PER_M = 1000;

function entityPerimeterMm(g: { kind: string; width?: number; height?: number; radius?: number; p1?: { x: number; y: number }; p2?: { x: number; y: number } }): number {
  if (g.kind === "rect" && g.width !== undefined && g.height !== undefined) {
    return 2 * (g.width + g.height);
  }
  if (g.kind === "circle" && g.radius !== undefined) {
    return 2 * Math.PI * g.radius;
  }
  if (g.kind === "line" && g.p1 && g.p2) {
    const dx = g.p2.x - g.p1.x;
    const dy = g.p2.y - g.p1.y;
    return Math.sqrt(dx * dx + dy * dy);
  }
  return 0;
}

export function estimateCutPerimeterMeters(ir: CadIr): number {
  let resolved;
  try {
    resolved = resolveIr(ir);
  } catch {
    return 0;
  }

  let perimMm = 0;
  for (const f of resolved.features) {
    if (f.suppressed) continue;
    if (f.kind === "extrude" || f.kind === "cut_extrude") {
      const sketch = resolved.sketches[f.profile];
      if (!sketch) continue;
      for (const g of sketch.geometry) {
        perimMm += entityPerimeterMm(g as never);
      }
    }
  }

  return perimMm / MM_PER_M;
}
```

```ts
// convex/cad/compile/machineCost.ts
import type { CadIr } from "../ir/types";
import { lookupProcess, type Process, BUILTIN_PROCESSES } from "./processes";
import { estimateCutPerimeterMeters } from "./perimeter";
import { estimatePartVolumeCm3 } from "./volume";

export interface MachineCostEntry {
  partId: string;
  process: string;
  setupUsd: number;
  cutUsd: number;
  removeUsd: number;
  buildUsd: number;
  bendUsd: number;
  lineTotalUsd: number;
}

export interface MachineCostResult {
  totalUsd: number;
  perPart: MachineCostEntry[];
}

function bendCount(ir: CadIr): number {
  let count = 0;
  for (const f of ir.features) {
    if (f.kind === "bend_flange" && !f.suppressed) count++;
  }
  return count;
}

function machineCostForPart(
  partId: string,
  partIr: CadIr,
  catalog: Record<string, Process>,
): MachineCostEntry | null {
  const process = lookupProcess(partIr.process, catalog);
  if (process.name === "none") return null;

  const setupUsd = process.setupUsd;
  const cutUsd = process.usdPerMeterCut * estimateCutPerimeterMeters(partIr);
  const buildUsd = process.usdPerCm3Built * estimatePartVolumeCm3(partIr);
  const bendUsd = process.usdPerBend * bendCount(partIr);
  // Phase 13 v0: removeUsd is 0 unless we model material stock dimensions; defer.
  const removeUsd = 0;

  const lineTotalUsd = setupUsd + cutUsd + removeUsd + buildUsd + bendUsd;

  return {
    partId,
    process: process.name,
    setupUsd, cutUsd, removeUsd, buildUsd, bendUsd, lineTotalUsd,
  };
}

export function compileMachineCost(
  ir: CadIr,
  catalog: Record<string, Process> = BUILTIN_PROCESSES,
): MachineCostResult {
  const perPart: MachineCostEntry[] = [];
  let totalUsd = 0;

  if (ir.parts && Object.keys(ir.parts).length > 0) {
    for (const [partId, part] of Object.entries(ir.parts)) {
      if ("kind" in part && part.kind === "external") continue;
      const partIr = (part as { ir: CadIr }).ir;
      const entry = machineCostForPart(partId, partIr, catalog);
      if (entry) {
        perPart.push(entry);
        totalUsd += entry.lineTotalUsd;
      }
    }
  } else {
    // Top-level CadIr is the part itself.
    const entry = machineCostForPart("root", ir, catalog);
    if (entry) {
      perPart.push(entry);
      totalUsd += entry.lineTotalUsd;
    }
  }

  return { totalUsd, perPart };
}
```
