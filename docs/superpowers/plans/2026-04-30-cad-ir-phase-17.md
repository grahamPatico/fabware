### Task 2: `mfg.cnc-min-internal-corner` rule

For Phase 17 v0, detect internal corners as `cut_extrude` rect sketches with `cornerRadius` field. If cornerRadius is undefined or < (toolDiameter / 2), the rule fires (the agent has to round internal corners to at least the tool radius).

Limitation: doesn't catch corners formed by intersecting cuts; only declared corner radii.

`convex/cad/validate/rules/cncMinInternalCorner.ts`:

```ts
import type { ResolvedIr } from "../../resolve/resolveIr";
import type { CadIr } from "../../ir/types";
import type { Violation } from "../../../plugins/types";

const DEFAULT_TOOL_D = 6.35; // 1/4" end mill

export function cncMinInternalCorner(resolved: ResolvedIr, original: CadIr): Violation[] {
  if (original.process !== "cnc") return [];
  const toolR = (original.cncToolDiameter ?? DEFAULT_TOOL_D) / 2;
  const out: Violation[] = [];

  for (const f of resolved.features) {
    if (f.kind !== "cut_extrude" || f.suppressed) continue;
    const sketch = resolved.sketches[f.profile];
    if (!sketch) continue;
    for (const g of sketch.geometry) {
      if (g.kind !== "rect") continue;
      const cr = g.cornerRadius;
      if (cr === undefined || cr < toolR) {
        out.push({
          ruleId: "mfg.cnc-min-internal-corner",
          severity: "error",
          message: `Pocket "${f.id}" has internal corner radius ${cr ?? 0}mm; CNC tool of diameter ${toolR * 2}mm requires ≥ ${toolR}mm.`,
          agentMessage: `The pocket cut by feature "${f.id}" has sharp internal corners that the ${toolR * 2}mm end mill can't reach. Either increase cornerRadius to at least ${toolR}mm, switch to a smaller tool (set cncToolDiameter), or change process to laser_cut.`,
          location: { kind: "feature", id: f.id },
        });
        break;
      }
    }
  }
  return out;
}
```

### Task 3: `mfg.cnc-pocket-too-deep` rule

A pocket's depth = `cut_extrude.distance`. Tool stiffness rule of thumb: depth ≤ 5× tool diameter without specialized long-reach tooling.

`convex/cad/validate/rules/cncPocketTooDeep.ts`:

```ts
import type { ResolvedIr } from "../../resolve/resolveIr";
import type { CadIr } from "../../ir/types";
import type { Violation } from "../../../plugins/types";

const DEFAULT_TOOL_D = 6.35;
const DEPTH_RATIO_LIMIT = 5;

export function cncPocketTooDeep(resolved: ResolvedIr, original: CadIr): Violation[] {
  if (original.process !== "cnc") return [];
  const toolD = original.cncToolDiameter ?? DEFAULT_TOOL_D;
  const out: Violation[] = [];

  for (const f of resolved.features) {
    if (f.kind !== "cut_extrude" || f.suppressed) continue;
    const depth = f.distance as number;
    if (depth > DEPTH_RATIO_LIMIT * toolD) {
      out.push({
        ruleId: "mfg.cnc-pocket-too-deep",
        severity: "warn",
        message: `Cut "${f.id}" depth ${depth}mm exceeds ${DEPTH_RATIO_LIMIT}× tool diameter (${DEPTH_RATIO_LIMIT * toolD}mm); tool deflection or chatter likely.`,
        agentMessage: `A ${depth}mm deep pocket is too aggressive for a ${toolD}mm end mill. Either use a longer/larger tool (cncToolDiameter), reduce the depth, or split into two passes from opposite faces.`,
        location: { kind: "feature", id: f.id },
      });
    }
  }
  return out;
}
```

3 tests.

Commit: `feat(cad-ir): mfg.cnc-pocket-too-deep — depth ≤ 5× tool diameter rule of thumb`
