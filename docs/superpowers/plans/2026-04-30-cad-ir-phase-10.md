// convex/cad/compile/cost.ts
import type { CadIr } from "../ir/types";
import { compileBom, type BomEntry } from "./bom";

export type PricingDb = Record<string, number>;

/** Tiny representative built-in. Caller supplies a richer map at the boundary. */
export const BUILTIN_PRICING: PricingDb = {
  "McMaster-Carr::91290A115": 0.42,   // M6 x 20 SHCS
  "McMaster-Carr::91290A130": 0.55,   // M6 x 30 SHCS
  "McMaster-Carr::91294A150": 0.75,   // M8 x 25 SHCS
  "McMaster-Carr::91100A030": 0.08,   // M3 hex nut
  "McMaster-Carr::91100A060": 0.15,   // M6 hex nut
  "McMaster-Carr::92141A012": 0.05,   // M6 flat washer
  "Misumi::B-6800ZZ": 3.20,            // 6800ZZ ball bearing
};

export interface CostEntry extends BomEntry {
  unitUsd: number;
  lineTotalUsd: number;
}

export interface CostResult {
  totalUsd: number;
  entries: CostEntry[];
  unknown: BomEntry[];
}

export function compileCost(ir: CadIr, db: PricingDb = BUILTIN_PRICING): CostResult {
  const bom = compileBom(ir);
  const entries: CostEntry[] = [];
  const unknown: BomEntry[] = [];
  let totalUsd = 0;

  for (const item of bom) {
    const key = `${item.vendor}::${item.partNumber}`;
    const unitUsd = db[key];
    if (unitUsd === undefined) {
      unknown.push(item);
      continue;
    }
    const lineTotalUsd = unitUsd * item.quantity;
    entries.push({ ...item, unitUsd, lineTotalUsd });
    totalUsd += lineTotalUsd;
  }

  return { totalUsd, entries, unknown };
}
```

// convex/cad/validate/rules/budgetExceeded.ts
import type { CadIr } from "../../ir/types";
import type { Violation } from "../../../plugins/types";
import { compileCost } from "../../compile/cost";

export function budgetExceeded(ir: CadIr): Violation[] {
  if (!ir.budget) return [];
  const cost = compileCost(ir);
  if (cost.totalUsd <= ir.budget.maxUsd) return [];
  return [{
    ruleId: "budget.exceeded",
    severity: "warn",
    message: `Assembly cost $${cost.totalUsd.toFixed(2)} exceeds declared budget $${ir.budget.maxUsd.toFixed(2)}.`,
    agentMessage: `The assembly's BOM costs $${cost.totalUsd.toFixed(2)} but the declared budget is $${ir.budget.maxUsd.toFixed(2)}. Either reduce part counts, swap to cheaper alternatives, or raise the budget.`,
  }];
}
```

import { applyPatch } from "../patch/apply";
import { resolveIr } from "../resolve/resolveIr";
import { validateSchemaTier } from "../validate/schemaTier";
import { validateConstraintTier } from "../validate/constraintTier";

import type { CadIr } from "../ir/types";
import { emptyIr } from "../ir/empty";

describe("CAD IR integration smoke test", () => {
  it("builds a full multi-part hinged enclosure with hardware, validates, and compiles every output", () => {
    // ── Body part: 80×60×3mm aluminum plate with two M6 mounting holes ──
    const bodyIr: CadIr = {
      ...emptyIr("mm"),
      parameters: {
        length: { id: "length", value: 80 },
        width: { id: "width", value: 60 },
        thickness: { id: "thickness", value: 3 },
        hole_d: { id: "hole_d", value: 6.5 },

          kind: "hole", id: "mount_holes", type: "simple",

      ],
    };

    const lidIr: CadIr = {
      ...emptyIr("mm"),
      parameters: {
        length: { id: "length", value: 80 },
        width: { id: "width", value: 60 },
        thickness: { id: "thickness", value: 1.5 },
      },
      sketches: {
        base: { id: "base", plane: "XY", geometry: [{ kind: "rect", id: "outer", center: { x: 0, y: 0 }, width: "length", height: "width", cornerRadius: 4 }] },

      ...emptyIr("mm"),
      budget: { maxUsd: 5.0 },
      parts: {
        body: { id: "body", ir: bodyIr, origin: { x: 0, y: 0, z: 0 } },
        lid: { id: "lid", ir: lidIr, origin: { x: 0, y: 0, z: 30 } }, // placed clearly above body

        limits: { min: 0, max: 90, unit: "deg" },

      const r = applyPatch(ir, {
        kind: "add_part",
        part: {
          id: `nut_${i}`, kind: "external",
          vendor: "McMaster-Carr", partNumber: "91100A060",
          description: "M6 hex nut",
          origin: { x: -30, y: 20, z: 50 + i },
          boundingBox: { width: 10, height: 10, depth: 5 },
        },
      });
      expect(r.schemaViolations).toEqual([]);
      ir = r.ir;
    }
    // Connect: bolt connection through body and lid
    {
      const r = applyPatch(ir, {
        kind: "add_connection",
        connection: { kind: "bolt", id: "bolt_main", spec: "M6", throughParts: ["body", "lid"], clearance: 0.5 },

      ir = r.ir;
    }
    // Connect each screw to its corresponding nut so externals are not "floating"
    for (let i = 0; i < 4; i++) {
      const r = applyPatch(ir, {
        kind: "add_connection",

    expect(mjcf).toContain("<joint name=\"lid_hinge\" type=\"hinge\"");

    const bom = compileBom(ir);
    expect(bom).toHaveLength(2);  // 1 SKU for screws (qty 4), 1 for nuts (qty 4)

    expect(hash1).toBe(hash2);
  });
