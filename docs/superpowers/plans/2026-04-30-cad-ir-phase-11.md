# CAD IR Phase 11 — Joint-range self-collision sampling

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development.

**Goal:** Sample each revolute and linear joint at N positions across its declared range and check inter-part AABB interference at each pose. Detects motion that physically can't happen — a hinge that slams the lid into the body at 30°, a slider that crashes into the housing at 80mm. Adds `assembly.joint-range-collision` to Tier 5.

**Why this slice:** Phase 4 ships joints with limits. Phase 8 ships static AABB interference. Combining them gives the agent kinematic feedback: "this hinge can't open past 45° without the lid hitting the wall." Without sampling, the agent only sees that the static at-rest pose is valid — it misses every other pose in the joint's range.

**Limitations:** AABB interference is conservative — rotation expands to a sphere envelope, which can over-flag. Sampling is finite (N=5 by default), so very sharp interference windows might be missed. Real swept-volume interference needs a geometry kernel; deferred. Also: the joint axis is treated as passing through the world origin; if your assembly places the joint elsewhere, conservative expansion catches it but the warning may be vague.

A joint pose is the offset/rotation that a child part takes at a given joint position. For revolute joints: a rotation about the joint axis. For linear joints: a translation along the axis. For fixed/ball/rigid_group: no sampling needed (fixed = always same pose; ball = unconstrained DOF, can't sample meaningfully; rigid_group = always rigid).

// convex/cad/geometry/jointPose.ts
import type { Joint, AxisRef, ParamRef } from "../ir/types";
import { transformBbox } from "./transform";
import type { AABB } from "./partBbox";

export interface JointPose {
  rotation: { rx: number; ry: number; rz: number };  // degrees
  translation: { x: number; y: number; z: number };   // mm or in (matches assembly units)
}

const RESTING: JointPose = {
  rotation: { rx: 0, ry: 0, rz: 0 },
  translation: { x: 0, y: 0, z: 0 },
};

function num(p: ParamRef | undefined, fallback = 0): number {
  if (typeof p === "number") return p;
  return fallback;
}

function axisVec(a: AxisRef): { x: number; y: number; z: number } {
  if (a.kind === "world") {
    if (a.axis === "x") return { x: 1, y: 0, z: 0 };
    if (a.axis === "y") return { x: 0, y: 1, z: 0 };
    return { x: 0, y: 0, z: 1 };
  }
  return { x: 0, y: 0, z: 1 }; // edge axis — conservative default
}

export function sampleJointPoses(j: Joint, n: number = 5): JointPose[] {
  if (j.kind === "fixed" || j.kind === "ball" || j.kind === "rigid_group") {
    return [RESTING];
  }
  if (!j.limits) return [RESTING];

  const samples: JointPose[] = [];
  const min = num(j.limits.min);
  const max = num(j.limits.max);
  if (n < 2) n = 2;
  const step = (max - min) / (n - 1);

  for (let i = 0; i < n; i++) {
    const v = min + step * i;
    if (j.kind === "revolute") {
      const factor = j.limits.unit === "rad" ? 180 / Math.PI : 1; // store in degrees
      const angleDeg = v * factor;
      const av = axisVec(j.axis);
      samples.push({
        rotation: { rx: angleDeg * av.x, ry: angleDeg * av.y, rz: angleDeg * av.z },
        translation: { x: 0, y: 0, z: 0 },
      });
    } else {
      // linear
      const factor = j.limits.unit === "in" ? 25.4 : 1; // store in mm
      const distance = v * factor;
      const av = axisVec(j.axis);
      samples.push({
        rotation: { rx: 0, ry: 0, rz: 0 },
        translation: { x: distance * av.x, y: distance * av.y, z: distance * av.z },
      });
    }
  }
  return samples;
}

export function applyPoseToBbox(bbox: AABB, pose: JointPose): AABB {
  return transformBbox(bbox, pose.translation, pose.rotation);
}
```

## Task 2: Tier 5 — jointRangeCollision rule

**Files:** `convex/cad/validate/rules/jointRangeCollision.ts`, `__tests__/rules-jointRangeCollision.test.ts`, `convex/cad/validate/assemblyTier.ts`

For each revolute/linear joint:
1. Sample N poses across its range
2. At each pose, compute the child part's transformed bbox
3. Check interference against all other parts' static bboxes
4. If any pose interferes (with anything other than the parent the joint connects), emit a violation

Skip if the child part has no bbox (empty IR).

// convex/cad/validate/rules/jointRangeCollision.ts
import type { CadIr, Joint, CadIr as _ } from "../../ir/types";
import type { Violation } from "../../../plugins/types";
import { computePartBbox, type AABB } from "../../geometry/partBbox";
import { transformBbox } from "../../geometry/transform";
import { sampleJointPoses, applyPoseToBbox } from "../../geometry/jointPose";

const N_SAMPLES = 5;

function aabbOverlap(a: AABB, b: AABB): boolean {
  if (a.maxX <= b.minX || b.maxX <= a.minX) return false;
  if (a.maxY <= b.minY || b.maxY <= a.minY) return false;
  if (a.maxZ <= b.minZ || b.maxZ <= a.minZ) return false;
  return true;
}

interface PartBbox { id: string; bbox: AABB | null; }

function getStaticBbox(part: NonNullable<CadIr["parts"]>[string]): AABB | null {
  let local: AABB | null;
  if ("kind" in part && part.kind === "external") {
    if (!part.boundingBox) return null;
    const bb = part.boundingBox as { width: number; height: number; depth: number };
    local = {
      minX: -bb.width / 2, maxX: bb.width / 2,
      minY: -bb.height / 2, maxY: bb.height / 2,
      minZ: 0, maxZ: bb.depth,
    };
  } else {
    local = computePartBbox((part as { ir: CadIr }).ir);
  }
  if (!local) return null;
  const origin = part.origin
    ? { x: part.origin.x as number, y: part.origin.y as number, z: part.origin.z as number }
    : undefined;
  const rotation = part.rotation
    ? { rx: part.rotation.rx as number, ry: part.rotation.ry as number, rz: part.rotation.rz as number }
    : undefined;
  return transformBbox(local, origin, rotation);
}

function getLocalBbox(part: NonNullable<CadIr["parts"]>[string]): AABB | null {
  if ("kind" in part && part.kind === "external") {
    if (!part.boundingBox) return null;
    const bb = part.boundingBox as { width: number; height: number; depth: number };
    return {
      minX: -bb.width / 2, maxX: bb.width / 2,
      minY: -bb.height / 2, maxY: bb.height / 2,
      minZ: 0, maxZ: bb.depth,
    };
  }
  return computePartBbox((part as { ir: CadIr }).ir);
}

export function jointRangeCollision(ir: CadIr): Violation[] {
  if (!ir.parts || !ir.joints) return [];
  const out: Violation[] = [];

  // Pre-compute every static bbox in the assembly
  const staticBboxes = new Map<string, AABB>();
  for (const [id, part] of Object.entries(ir.parts)) {
    const bb = getStaticBbox(part);
    if (bb) staticBboxes.set(id, bb);
  }

  for (const j of ir.joints) {
    if (j.kind === "fixed" || j.kind === "ball" || j.kind === "rigid_group") continue;
    if (!j.limits) continue;

    const childPart = ir.parts[j.b];
    if (!childPart) continue;
    const childLocal = getLocalBbox(childPart);
    if (!childLocal) continue;

    // Apply child's static origin to local bbox first, then sample poses
    const childOrigin = childPart.origin
      ? { x: childPart.origin.x as number, y: childPart.origin.y as number, z: childPart.origin.z as number }
      : undefined;
    const childStaticBeforePose = transformBbox(childLocal, childOrigin, undefined);

    const poses = sampleJointPoses(j, N_SAMPLES);
    let collisionPose: number | null = null;
    let collidingPartId: string | null = null;

    for (let i = 0; i < poses.length; i++) {
      const childAtPose = applyPoseToBbox(childStaticBeforePose, poses[i]);
      for (const [otherId, otherBbox] of staticBboxes) {
        if (otherId === j.a || otherId === j.b) continue; // skip joint endpoints
        if (aabbOverlap(childAtPose, otherBbox)) {
          collisionPose = i;
          collidingPartId = otherId;
          break;
        }
      }
      if (collidingPartId) break;
    }

    if (collidingPartId !== null) {
      out.push({
        ruleId: "assembly.joint-range-collision",
        severity: "warn",
        message: `Joint "${j.id}" sweeps part "${j.b}" into part "${collidingPartId}" at sample ${collisionPose}/${poses.length - 1}.`,
        agentMessage: `When joint "${j.id}" moves through its range, part "${j.b}" hits "${collidingPartId}". Reduce the joint's range, move the obstacle, or change the parts' geometries so the swept volume is clear.`,
        location: { kind: "feature", id: j.id },
      });
    }
  }

  return out;
}
```

## Out of scope for Phase 11 (deferred)

- **Real swept-volume interference** — needs OCCT-level operations; defer to a phase with sandbox geometry.
- **Joint axis location** — currently treated as passing through world origin; real axis location requires geometry.
- **Higher sampling rates / adaptive sampling** — could miss narrow interference windows; future tuning.
- **Combined joint sweeps** — multi-DOF assemblies (revolute + linear simultaneously) sample each joint independently; future phase if needed.
